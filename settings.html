// ============================================ // ===== НАСТРОЙКИ ===== // ============================================ function renderSettings() { const nameEl = document.getElementById('settingsName'); if (nameEl) nameEl.textContent = APP.user.name || '—'; const themeSelect = document.getElementById('themeSelect'); if (themeSelect) themeSelect.value = APP.settings.theme || 'light'; } // Изменение имени const editProfileBtn = document.getElementById('editProfileBtn'); if (editProfileBtn) { editProfileBtn.addEventListener('click', function() { const newName = prompt('Введите ваше имя:', APP.user.name); if (newName && newName.trim()) { APP.user.name = newName.trim(); saveAppData(); renderSettings(); alert('✅ Имя обновлено!'); } }); } // Смена темы const themeSelect = document.getElementById('themeSelect'); if (themeSelect) { themeSelect.addEventListener('change', function() { APP.settings.theme = this.value; saveAppData(); document.body.classList.toggle('dark', this.value === 'dark'); }); } // Очистка кэша const clearCacheBtn = document.getElementById('clearCacheBtn'); if (clearCacheBtn) { clearCacheBtn.addEventListener('click', function() { if (confirm('Очистить кэш? (Ваши викторины и результаты не удалятся)')) { if ('caches' in window) { caches.keys().then(keys => keys.forEach(key => caches.delete(key))); } alert('✅ Кэш очищен'); } }); } // Запуск при загрузке страницы настроек document.addEventListener('DOMContentLoaded', () => { if (document.getElementById('page-settings')) { renderSettings(); } });
